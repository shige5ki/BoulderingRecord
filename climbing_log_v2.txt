<script>
const KEY = 'climbing-log-v2';
const defaultSettings = { gyms: ['B-PUMP TOKYO', 'NOBOROCK'], grades: ['3段','2段','初段','1級','2級','3級','4級','5級','6級','7級'] };

let app = loadData();

function loadData() {
  try {
    let d = JSON.parse(localStorage.getItem(KEY));
    if (!d || !d.users) throw 'No Data';
    return d;
  } catch(e) {
    return {
      users: ['自分'], currentUser: '自分',
      data: { '自分': { gyms: [...defaultSettings.gyms], grades: [...defaultSettings.grades], sessions: [], tasks: [], activeSession: null } }
    };
  }
}
function save() { localStorage.setItem(KEY, JSON.stringify(app)); }
function getU() { return app.data[app.currentUser]; }
function esc(s){ return String(s??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])) }

function init() {
  document.getElementById('tDate').value = document.getElementById('mDate').value = new Date().toLocaleDateString('sv-SE');
  renderUserSelect();
  renderAll();
}

document.querySelectorAll('.tab').forEach(b=>b.addEventListener('click', e => {
  document.querySelectorAll('.tab').forEach(t=>t.classList.remove('active'));
  document.querySelectorAll('.panel').forEach(p=>p.classList.remove('active'));
  e.target.classList.add('active');
  document.getElementById(e.target.dataset.tab).classList.add('active');
  if(e.target.dataset.tab === 'history') renderHistory();
  if(e.target.dataset.tab === 'analytics') renderAnalytics();
}));

function renderUserSelect() {
  const sel = document.getElementById('userSelect');
  sel.innerHTML = app.users.map(u => `<option ${u===app.currentUser?'selected':''}>${esc(u)}</option>`).join('');
}
document.getElementById('userSelect').onchange = (e) => {
  app.currentUser = e.target.value;
  save(); renderAll();
};
document.getElementById('addUser').onclick = () => {
  let u = document.getElementById('newUserName').value.trim();
  if(!u || app.users.includes(u)) return;
  app.users.push(u);
  app.data[u] = { gyms: [...defaultSettings.gyms], grades: [...defaultSettings.grades], sessions: [], tasks: [], activeSession: null };
  app.currentUser = u;
  document.getElementById('newUserName').value = '';
  save(); renderUserSelect(); renderAll();
};

function renderAll() {
  renderGyms(); renderGrades(); checkinStateUI();
}

function renderGyms() {
  const u = getU();
  const opts = u.gyms.map(g => `<option>${esc(g)}</option>`).join('');
  document.getElementById('ciGym').innerHTML = document.getElementById('mGym').innerHTML = opts;
  document.getElementById('gymList').innerHTML = u.gyms.map((g,i) => `<div class="inline" style="margin:7px 0"><span>${esc(g)}</span><button class="secondary small danger" onclick="delGym(${i})">削除</button></div>`).join('');
}
function renderGrades() {
  const u = getU();
  document.getElementById('tGrade').innerHTML = u.grades.map(g => `<option>${esc(g)}</option>`).join('');
  document.getElementById('gradeList').innerHTML = u.grades.map((g,i) => `<div class="inline" style="margin:7px 0"><span>${esc(g)}</span><button class="secondary small danger" onclick="delGrade(${i})">削除</button></div>`).join('');
}

document.getElementById('addGym').onclick=()=>{let v=document.getElementById('newGym').value.trim(); if(v){ getU().gyms.push(v); save(); document.getElementById('newGym').value=''; renderGyms();} };
document.getElementById('addGrade').onclick=()=>{let v=document.getElementById('newGrade').value.trim(); if(v){ getU().grades.push(v); save(); document.getElementById('newGrade').value=''; renderGrades();} };
function delGym(i){ getU().gyms.splice(i,1); save(); renderGyms(); }
function delGrade(i){ getU().grades.splice(i,1); save(); renderGrades(); }

function checkinStateUI() {
  const as = getU().activeSession;
  if(as) {
    document.getElementById('checkinCard').style.display = 'none';
    document.getElementById('activeSessionBox').style.display = 'block';
    document.getElementById('asGym').textContent = as.gym;
    document.getElementById('asStart').textContent = as.start;
  } else {
    document.getElementById('checkinCard').style.display = 'block';
    document.getElementById('activeSessionBox').style.display = 'none';
  }
}

document.getElementById('btnCheckin').onclick = () => {
  const now = new Date();
  getU().activeSession = { id: Date.now(), gym: document.getElementById('ciGym').value, date: now.toLocaleDateString('sv-SE'), start: now.toLocaleTimeString('ja-JP',{hour:'2-digit',minute:'2-digit'}) };
  save(); checkinStateUI();
};

document.getElementById('btnCheckout').onclick = () => {
  const u = getU();
  if(!u.activeSession) return;
  const now = new Date();
  let end = now.toLocaleTimeString('ja-JP',{hour:'2-digit',minute:'2-digit'});
  let s = {...u.activeSession, end, duration: calcDuration(u.activeSession.start, end)};
  u.sessions.push(s);
  u.activeSession = null;
  save(); checkinStateUI();
};

document.getElementById('btnSaveManualSession').onclick = () => {
  let s = { id: Date.now(), gym: document.getElementById('mGym').value, date: document.getElementById('mDate').value, start: document.getElementById('mStart').value, end: document.getElementById('mEnd').value };
  s.duration = calcDuration(s.start, s.end);
  getU().sessions.push(s); save();
  document.getElementById('manualSessionCard').style.display = 'none';
  document.querySelector('#checkinCard .secondary').style.display = 'inline-block';
};

function calcDuration(start, end) {
  if(!start || !end) return '';
  let [ah,am]=start.split(':').map(Number), [bh,bm]=end.split(':').map(Number);
  let mins = (bh*60+bm) - (ah*60+am);
  if(mins<0) mins+=1440;
  return `${Math.floor(mins/60)}h ${mins%60}m`;
}

function saveTask(clone=false) {
  const u = getU();
  let t = {
    id: Date.now(),
    date: document.getElementById('tDate').value,
    sessionId: u.activeSession ? u.activeSession.id : null,
    name: document.getElementById('tName').value.trim(),
    grade: document.getElementById('tGrade').value,
    wall: document.getElementById('tWall').value,
    result: document.getElementById('tResult').value,
    note: document.getElementById('tNote').value.trim()
  };
  u.tasks.push(t); save();
  
  if(!clone) {
    document.getElementById('tName').value = '';
    document.getElementById('tNote').value = '';
    document.getElementById('tResult').value = 'send';
  }
  let msg = document.createElement('div');
  msg.textContent = '✅ 保存しました！';
  msg.style = 'color:var(--success);font-weight:bold;text-align:center;margin-top:10px';
  document.getElementById('tNote').parentNode.appendChild(msg);
  setTimeout(()=>msg.remove(), 2000);
}
document.getElementById('btnSaveTask').onclick = () => saveTask(false);
document.getElementById('btnSaveCloneTask').onclick = () => saveTask(true);

function renderHistory() {
  const u = getU();
  const list = document.getElementById('historyList');
  
  let items = [...u.sessions.map(s=>({...s, type:'session'})), ...u.tasks.map(t=>({...t, type:'task'}))];
  items.sort((a,b) => b.date.localeCompare(a.date) || b.id - a.id);
  
  if(items.length===0){ list.innerHTML='<div style="text-align:center;padding:20px;color:var(--muted)">記録がありません</div>'; return; }
  
  let html = '', curDate = '';
  items.forEach(item => {
    if(item.date !== curDate) {
      curDate = item.date;
      html += `<div class="history-date">${curDate.replace(/-/g, '/')}</div>`;
    }
    if(item.type === 'session') {
      html += `<div class="card h-item" style="border-left:4px solid var(--accent)">
        <div class="inline"><strong>🏢 ${esc(item.gym)}</strong><span class="muted">${item.start}〜${item.end} (${item.duration})</span></div>
        <div style="margin-top:8px;text-align:right">
          <button class="secondary small danger" onclick="delItem('session',${item.id})">削除</button>
        </div>
      </div>`;
    } else {
      let resBadge = item.result==='flash' ? '<span class="pill flash">⚡1撃</span>' : item.result==='send' ? '<span class="pill send">○完登</span>' : '<span class="pill fail">×未完登</span>';
      html += `<div class="card h-item">
        <div class="inline"><strong>${esc(item.name)||'課題'}</strong><span class="pill">${esc(item.grade)}</span>${resBadge}</div>
        <div class="muted" style="margin-top:4px">${esc(item.wall)} ${item.note ? ' ／ '+esc(item.note) : ''}</div>
        <div style="margin-top:8px;text-align:right">
          <button class="secondary small danger" onclick="delItem('task',${item.id})">削除</button>
        </div>
      </div>`;
    }
  });
  list.innerHTML = html;
}

function delItem(type, id) {
  const u = getU();
  if(type==='session') u.sessions = u.sessions.filter(s=>s.id!==id);
  else u.tasks = u.tasks.filter(t=>t.id!==id);
  save(); renderHistory();
}

function renderAnalytics() {
  const u = getU();
  const today = new Date();
  const y = today.getFullYear(), m = today.getMonth();
  document.getElementById('calMonth').textContent = `${y}年 ${m+1}月`;
  
  const firstDay = new Date(y, m, 1).getDay();
  const days = new Date(y, m+1, 0).getDate();
  const cal = document.getElementById('calendar');
  let cHtml = ['日','月','火','水','木','金','土'].map(d=>`<div class="cal-head">${d}</div>`).join('');
  
  for(let i=0; i<firstDay; i++) cHtml += `<div></div>`;
  
  let mPrefix = `${y}-${String(m+1).padStart(2,'0')}`;
  let activeDates = new Set([...u.sessions, ...u.tasks].filter(x => x.date.startsWith(mPrefix)).map(x => x.date));
  
  for(let d=1; d<=days; d++) {
    let dateStr = `${mPrefix}-${String(d).padStart(2,'0')}`;
    let cls = activeDates.has(dateStr) ? 'cal-day active' : 'cal-day';
    cHtml += `<div class="${cls}">${d}</div>`;
  }
  cal.innerHTML = cHtml;

  let tSend = u.tasks.filter(t=>t.result==='send'||t.result==='flash').length;
  document.getElementById('statSend').textContent = tSend;
  document.getElementById('statRate').textContent = u.tasks.length ? Math.round((tSend/u.tasks.length)*100)+'%' : '0%';
  
  let mSessions = u.sessions.filter(s=>s.date.startsWith(mPrefix));
  let mins = mSessions.reduce((acc, s) => {
    let mMatch = s.duration ? s.duration.match(/(\d+)h\s*(\d+)m/) : null;
    if(mMatch) return acc + parseInt(mMatch[1])*60 + parseInt(mMatch[2]);
    return acc;
  }, 0);
  document.getElementById('statHours').textContent = (mins/60).toFixed(1);

  let maxGradeIdx = Infinity, maxGradeStr = 'なし';
  u.tasks.filter(t=>t.result==='send'||t.result==='flash').forEach(t => {
    let idx = u.grades.indexOf(t.grade);
    if(idx !== -1 && idx < maxGradeIdx) { maxGradeIdx = idx; maxGradeStr = t.grade; }
  });
  document.getElementById('statMax').textContent = maxGradeStr;

  let gCount = {}; u.tasks.forEach(t => { if(t.result==='send'||t.result==='flash') gCount[t.grade] = (gCount[t.grade]||0)+1; });
  let gMax = Math.max(1, ...Object.values(gCount));
  document.getElementById('chartGrade').innerHTML = u.grades.filter(g=>gCount[g]).map(g => renderBar(g, gCount[g], gMax)).join('') || '<div class="muted">データなし</div>';

  let wCount = {}; u.tasks.forEach(t => wCount[t.wall] = (wCount[t.wall]||0)+1);
  let wMax = Math.max(1, ...Object.values(wCount));
  document.getElementById('chartWall').innerHTML = Object.entries(wCount).sort((a,b)=>b[1]-a[1]).map(([w,c]) => renderBar(w, c, wMax)).join('') || '<div class="muted">データなし</div>';
}

function renderBar(label, count, max) {
  let pct = Math.round((count/max)*100);
  return `<div class="bar-row"><div class="bar-label">${esc(label)}</div><div class="bar-track"><div class="bar-fill" style="width:${pct}%"></div></div><div class="muted" style="width:20px">${count}</div></div>`;
}
